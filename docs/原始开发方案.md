## 一、项目概述

一个以原神为主题的综合性功能站点，包含四大视图：**首页画廊 / 下载 / 功能 / 关于**。前后端分离，前端负责视觉与交互，后端负责安全中转 Uptime Kuma API 数据，并支持通过管理后台动态配置站点内容与外观。

**核心目标**：
- 美观度拉到最高（参考原神官网 + 高质量社区仓库）
- 安全性不妥协（API Key、CSP、速率限制、输入验证）
- 手机端完整可用
- 所有内容与外观可通过后端管理后台动态调整，无需修改代码

### 4.5 数据合并逻辑

Kuma 的 heartbeat 接口返回的数据**不包含监控名称**，后端需要：

1. 调用 `GET /api/status-page/{slug}` → 获取监控列表（含名称）
2. 调用 `GET /api/status-page/heartbeat/{slug}` → 获取心跳数据
3. 以 `monitorID` 为键做 merge，输出统一格式

### 4.6 缓存与降级

| 策略 | 说明 |
|---|---|
| REST 结果缓存 | `node-cache` 缓存 30~60 秒 |
| 缓存过期返回旧数据 | Kuma 不可用时返回最后一次成功数据 + 标记 `stale: true` |
| 前端骨架屏 | 加载中显示占位卡片 |
| 失败重试 | 请求失败显示「重试」按钮，不自动狂刷 |

### 4.7 Kuma 认证

**只用 API Key，不用用户名密码**。以 Basic Auth 发送，username 留空，password 填 Key。Key 设置作用域限制 + 过期时间 + 定期轮换。

#### 4.8.3 配置服务实现

```javascript
// services/configService.js
const fs = require('fs').promises;
const path = require('path');

const CONFIG_PATH = path.join(__dirname, '../data/config.json');

async function getConfig() {
  const data = await fs.readFile(CONFIG_PATH, 'utf-8');
  return JSON.parse(data);
}

async function getConfigSection(section) {
  const config = await getConfig();
  return config[section] || null;
}

async function updateConfigSection(section, data) {
  const config = await getConfig();
  config[section] = { ...config[section], ...data };
  await fs.writeFile(CONFIG_PATH, JSON.stringify(config, null, 2));
  return config[section];
}

module.exports = { getConfig, getConfigSection, updateConfigSection };
```

#### 4.8.4 管理员认证

```javascript
// middleware/auth.js
const jwt = require('jsonwebtoken');

function requireAdmin(req, res, next) {
  const token = req.headers.authorization?.replace('Bearer ', '');
  if (!token) return res.status(401).json({ error: '未提供认证令牌' });

  try {
    const decoded = jwt.verify(token, process.env.JWT_SECRET);
    if (!decoded.isAdmin) return res.status(403).json({ error: '权限不足' });
    req.user = decoded;
    next();
  } catch (err) {
    return res.status(401).json({ error: '令牌无效或已过期' });
  }
}

module.exports = { requireAdmin };
```

登录接口：

```javascript
// routes/auth.js
const router = require('express').Router();
const jwt = require('jsonwebtoken');

router.post('/login', async (req, res) => {
  const { username, password } = req.body;

  if (username !== process.env.ADMIN_USERNAME || password !== process.env.ADMIN_PASSWORD) {
    return res.status(401).json({ error: '用户名或密码错误' });
  }

  const token = jwt.sign(
    { username, isAdmin: true },
    process.env.JWT_SECRET,
    { expiresIn: '24h' }
  );

  res.json({ token });
});

module.exports = router;
```

#### 5.7.6 中文回退策略

## 阶段 2：Kuma REST API 集成

### 目标

通过后端调用 Uptime Kuma 的公开 REST 接口，获取状态页配置和心跳数据，合并后返回给前端。

### 关键 API 端点

| 端点 | 认证 | 返回内容 |
|---|---|---|
| `/api/status-page/{slug}` | 公开 | 状态页配置、监控列表、事件 |
| `/api/status-page/heartbeat/{slug}` | 公开 | 最新 100 条心跳 + 24h 可用率 |

### 服务层（`services/kumaRest.js`）

```javascript
const axios = require('axios');

const KUMA_URL = process.env.KUMA_URL;
const SLUG = process.env.KUMA_STATUS_SLUG;

const kumaClient = axios.create({
  baseURL: KUMA_URL,
  timeout: 10000,
  headers: { 'Content-Type': 'application/json' },
});

async function fetchStatusPage() {
  const { data } = await kumaClient.get(`/api/status-page/${SLUG}`);
  return data;
}

async function fetchHeartbeat() {
  const { data } = await kumaClient.get(`/api/status-page/heartbeat/${SLUG}`);
  return data;
}

async function getMergedMonitors() {
  const [statusPage, heartbeat] = await Promise.all([
    fetchStatusPage(),
    fetchHeartbeat(),
  ]);

  const heartbeatMap = {};
  heartbeat.heartbeatList.forEach((hb) => {
    heartbeatMap[hb.monitorID] = hb;
  });

  const monitors = statusPage.publicGroupList.flatMap((group) =>
    group.monitorList.map((monitor) => {
      const hb = heartbeatMap[monitor.id] || {};
      return {
        id: monitor.id,
        name: monitor.name,
        type: monitor.type,
        status: hb.status === 1 ? 'up' : hb.status === 0 ? 'down' : 'unknown',
        uptime24h: heartbeat.uptimeList?.[`${monitor.id}_24`] ?? null,
        ping: hb.ping ?? null,
        lastHeartbeat: hb.time ?? null,
      };
    })
  );

  const summary = {
    total: monitors.length,
    up: monitors.filter((m) => m.status === 'up').length,
    down: monitors.filter((m) => m.status === 'down').length,
    maintenance: 0,
  };

  return { monitors, summary, lastUpdated: new Date().toISOString() };
}

module.exports = { getMergedMonitors };
```

### 路由层（`routes/status.js`）

```javascript
const router = require('express').Router();
const cache = require('../cache');
const { getMergedMonitors } = require('../services/kumaRest');

router.get('/monitors', async (req, res) => {
  try {
    let data = cache.get('monitors');
    if (!data) {
      data = await getMergedMonitors();
      cache.set('monitors', data, process.env.CACHE_TTL || 30);
    }
    res.json(data);
  } catch (err) {
    const stale = cache.get('monitors');
    if (stale) {
      return res.json({ ...stale, stale: true });
    }
    res.status(503).json({ error: 'Kuma 服务暂时不可用' });
  }
});

router.get('/summary', async (req, res) => {
  const data = cache.get('monitors');
  if (!data) return res.status(503).json({ error: '数据尚未就绪' });
  res.json(data.summary);
});

module.exports = router;
```

### 缓存层（`cache.js`）

```javascript

```

```javascript
}
```

### 管理员认证（`middleware/auth.js` + `routes/auth.js`）

```javascript
// middleware/auth.js
const jwt = require('jsonwebtoken');

function requireAdmin(req, res, next) {
  const token = req.headers.authorization?.replace('Bearer ', '');
  if (!token) return res.status(401).json({ error: '未提供认证令牌' });

  try {
    const decoded = jwt.verify(token, process.env.JWT_SECRET);
    if (!decoded.isAdmin) return res.status(403).json({ error: '权限不足' });
    req.user = decoded;
    next();
  } catch (err) {
    return res.status(401).json({ error: '令牌无效或已过期' });
  }
}

module.exports = { requireAdmin };
```

```javascript
// routes/auth.js
const router = require('express').Router();
const jwt = require('jsonwebtoken');

router.post('/login', async (req, res) => {
  const { username, password } = req.body;

  if (username !== process.env.ADMIN_USERNAME || password !== process.env.ADMIN_PASSWORD) {
    return res.status(401).json({ error: '用户名或密码错误' });
  }

  const token = jwt.sign(
    { username, isAdmin: true },
    process.env.JWT_SECRET,
    { expiresIn: '24h' }
  );

  res.json({ token });
});
```

```javascript
class Gallery {
  constructor(container) {
    this.container = container;
    this.slides = container.querySelectorAll('.gallery-slide');
    this.currentIndex = 0;
    this.isScrolling = false;
    this.init();
  }

  init() {
    const dotsContainer = document.getElementById('galleryDots');
    this.slides.forEach((_, i) => {
      const dot = document.createElement('button');
      dot.className = 'gallery-dot' + (i === 0 ? ' active' : '');
      dot.addEventListener('click', () => this.goTo(i));
      dotsContainer.appendChild(dot);
    });

    this.container.addEventListener('wheel', (e) => {
      e.preventDefault();
      if (this.isScrolling) return;
      const dir = e.deltaY > 0 ? 1 : -1;
      this.goTo(this.currentIndex + dir);
    }, { passive: false });

    this.initTouch();
    this.animateSlide(0);
  }

  initTouch() {
    let startY = 0;
    this.container.addEventListener('touchstart', (e) => {
      startY = e.touches[0].clientY;
    }, { passive: true });

    this.container.addEventListener('touchend', (e) => {
      const endY = e.changedTouches[0].clientY;
      const diff = startY - endY;
      if (Math.abs(diff) > 50) {
        const dir = diff > 0 ? 1 : -1;
        this.goTo(this.currentIndex + dir);
      }
    }, { passive: true });
  }

  goTo(index) {
    if (index < 0 || index >= this.slides.length) return;
    if (index === this.currentIndex) return;

    this.isScrolling = true;
    this.currentIndex = index;

    this.container.scrollTo({
      top: index * this.container.clientHeight,
      behavior: 'smooth',
    });

    document.querySelectorAll('.gallery-dot').forEach((dot, i) => {
      dot.classList.toggle('active', i === index);
    });

    this.animateSlide(index);

    setTimeout(() => { this.isScrolling = false; }, 800);
  }

  animateSlide(index) {
    const slide = this.slides[index];
    const title = slide.querySelector('.hero-title, .slide-title');
    if (title && title.dataset.text) this.animateTitle(title);

    slide.querySelectorAll('[data-animate]').forEach((el) => {
      el.style.animation = 'none';
      el.offsetHeight;
      el.style.animation = '';
    });
  }

  animateTitle(el) {
    const text = el.dataset.text;
    el.textContent = '';
    [...text].forEach((char, i) => {
      const span = document.createElement('span');
      span.textContent = char;
      span.style.animationDelay = `${i * 0.06}s`;
      span.className = 'char';
      el.appendChild(span);
    });
  }
}
```

```javascript
import * as THREE from 'three';

export function createStarfield(canvas) {
  const scene = new THREE.Scene();
  const camera = new THREE.PerspectiveCamera(75, canvas.clientWidth / canvas.clientHeight, 0.1, 1000);
  camera.position.z = 5;

  const renderer = new THREE.WebGLRenderer({ canvas, alpha: true, antialias: true });
  renderer.setSize(canvas.clientWidth, canvas.clientHeight);
  renderer.setPixelRatio(Math.min(window.devicePixelRatio, 2));

  const count = window.innerWidth < 768 ? 4000 : 10000;
  const positions = new Float32Array(count * 3);
  const sizes = new Float32Array(count);
  const phases = new Float32Array(count);

  for (let i = 0; i < count; i++) {
    positions[i * 3] = (Math.random() - 0.5) * 40;
    positions[i * 3 + 1] = (Math.random() - 0.5) * 25;
    positions[i * 3 + 2] = (Math.random() - 0.5) * 30 - 5;
    sizes[i] = Math.random() * 2 + 0.5;
    phases[i] = Math.random() * Math.PI * 2;
  }

  const geometry = new THREE.BufferGeometry();
  geometry.setAttribute('position', new THREE.BufferAttribute(positions, 3));
  geometry.setAttribute('size', new THREE.BufferAttribute(sizes, 1));
  geometry.setAttribute('phase', new THREE.BufferAttribute(phases, 1));

  const material = new THREE.ShaderMaterial({
    uniforms: {
      uTime: { value: 0 },
      uColor: { value: new THREE.Color('#e8c877') },
    },
    vertexShader: `
      attribute float size;
      attribute float phase;
      varying float vPhase;
      void main() {
        vPhase = phase;
        vec4 mvPosition = modelViewMatrix * vec4(position, 1.0);
        gl_PointSize = size * (300.0 / -mvPosition.z);
        gl_Position = projectionMatrix * mvPosition;
      }
    `,
    fragmentShader: `
      uniform float uTime;
      uniform vec3 uColor;
      varying float vPhase;
      void main() {
        float twinkle = sin(uTime * 2.0 + vPhase) * 0.5 + 0.5;
        float alpha = twinkle * 0.8 + 0.2;
        float dist = length(gl_PointCoord - vec2(0.5));
        if (dist > 0.5) discard;
        float glow = 1.0 - dist * 2.0;
        gl_FragColor = vec4(uColor, alpha * glow);
      }
    `,
    transparent: true,
    depthWrite: false,
    blending: THREE.AdditiveBlending,
  });

  const stars = new THREE.Points(geometry, material);
  scene.add(stars);

  let animId;
  function animate(time) {
    material.uniforms.uTime.value = time * 0.001;
    stars.rotation.y = time * 0.00002;
    renderer.render(scene, camera);
    animId = requestAnimationFrame(animate);
  }
  animId = requestAnimationFrame(animate);

  document.addEventListener('visibilitychange', () => {
    if (document.hidden) cancelAnimationFrame(animId);
    else animId = requestAnimationFrame(animate);
  });

  window.addEventListener('resize', () => {
    camera.aspect = canvas.clientWidth / canvas.clientHeight;
    camera.updateProjectionMatrix();
    renderer.setSize(canvas.clientWidth, canvas.clientHeight);
  });

  return { scene, renderer, animId };
}
```

### 性能优化清单

|---|---|
| 只用 transform/opacity | 避免 left/top/width/height 触发重排 |

| 粒子数量自适应 | 已在星空代码中按屏宽调整 |
| prefers-reduced-motion | 见下方 |
| visibilitychange 暂停 | 星空代码已实现 |

## 阶段 10：（可选）Socket.IO 实时推送

### 目标

通过后端与 Kuma 建立 Socket.IO 长连接，实时推送状态变化给前端。

### 后端连接管理（`services/kumaSocket.js`）

```javascript
const { io } = require('socket.io-client');

let socket = null;
let listeners = [];

function connectKuma() {
  socket = io(process.env.KUMA_URL, {
    transports: ['websocket'],
    reconnection: true,
    reconnectionDelay: 5000,
  });

  socket.on('connect', () => {
    socket.emit('login', {
      username: process.env.KUMA_USERNAME,
      password: process.env.KUMA_PASSWORD,
    });
  });

  socket.on('loginSuccess', () => {
    console.log('Kuma 已连接');
    socket.emit('getMonitorList');
  });

  socket.on('heartbeat', (data) => {
    listeners.forEach((fn) => fn(data));
  });

  socket.on('disconnect', () => {
    console.log('Kuma 连接断开，自动重连中...');
  });
}

function onHeartbeat(fn) {
  listeners.push(fn);
}

module.exports = { connectKuma, onHeartbeat };
```

### 推送给前端（SSE）

```javascript
router.get('/events', (req, res) => {
  res.writeHead(200, {
    'Content-Type': 'text/event-stream',
    'Cache-Control': 'no-cache',
    Connection: 'keep-alive',
  });

  onHeartbeat((data) => {
    res.write(`data: ${JSON.stringify(data)}\n\n`);
  });
});
```

### 前端监听

```javascript
const evtSource = new EventSource(`${API_BASE}/events`);
evtSource.onmessage = (e) => {
  const data = JSON.parse(e.data);
  updateMonitorCard(data);
};
```

### 验收标准

- Kuma 状态变化后 5 秒内前端卡片更新
- 后端断开重连后自动恢复
- 页面关闭时 SSE 连接自动断开
